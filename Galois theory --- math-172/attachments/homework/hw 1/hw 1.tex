\begin{problem}[1.]
	Prove the following identities in an arbitrary ring $R$.
	\begin{enumerate}
		\item $0 a = 0$ 
		\item $-a = (-1) a$ 
		\item $(-a) b = - (ab)$
	\end{enumerate}
\end{problem}

\pagebreak

\begin{problem}[2.]
	Describe explicitly the smallest subring of the complex numbers which contains the real cube root.
\end{problem}

\pagebreak

\begin{problem}[3.]
	Prove that $7 + \sqrt[3]{2}$ and $\sqrt{3} + \sqrt{-5}$ are algebraic numbers.
\end{problem}

\pagebreak

\begin{problem}[4.]
	Let $\mathbb{Q}[\alpha, \beta]$ denote the smallest subring of $\mathbb{C}$ containing the rational numbers $\mathbb{Q}$ and the elements $\alpha = \sqrt{2}$ and $\beta = \sqrt{3}$. Let $\gamma = \alpha + \beta$. Is $\mathbb{Q}[\alpha, \beta] = \mathbb{Q}[\gamma]$? Is $\mathbb{Z}[\alpha, \beta] = \mathbb{Z}[\gamma]$?
\end{problem}

\pagebreak

\begin{problem}[5.]
	Decide whether or not $S$ is a subring of $R$ when $S$ is the set of all rational numbers $a / b$ where $b$ is not divisible by $3$ and $R = \mathbb{Q}$.
\end{problem}

\pagebreak

\begin{problem}[6.]
	Determine the units in
	\begin{enumerate}
		\item $\mathbb{Z} / 12$ 
		\item $\mathbb{Z} / 8$
		\item $\mathbb{Z} / n$.
	\end{enumerate}
\end{problem}

\pagebreak

\begin{problem}[7.]
	Let $\mathbb{F}$ be a field. Prove that the set of all formal power series $\mathbb{F}[[t]]$ is a ring and determine the units in it.
\end{problem}

\pagebreak

\begin{problem}[8.]
	Let $R$ be a ring of prime characteristic $p$. Prove that the map $R \to R$ defined by $x \mapsto x^{p}$ is a ring homomorphism.
\end{problem}

\pagebreak

\begin{problem}[9.]
	An element $x$ of a ring $R$ is called nilpotent if some power is zero. Prove that if $x$ is nilpotent, then $1 + x$ is a unit.
\end{problem}

\pagebreak

\begin{problem}[10.]
	Identify the following ring: $\mathbb{Z}[i] / (2 + i)$.
\end{problem}

\pagebreak

\begin{problem}[11.]
	Determine the structure of the rings
	\begin{enumerate}
		\item $\mathbb{Z}[x] / (x^{2} + 3, 5)$ 
		\item $\mathbb{Z}[x] / (x^{2} + 3, 3)$
	\end{enumerate}
\end{problem}
